# Python 3.12 specifically: harbor 0.20.0 is built against it, and the harness
# uses tomllib (3.11+). 3.13+ is untested here.

# This file covers what runs ON YOUR MACHINE: the Harbor runner, the OAuth
# bridge, and the harness scripts. The graders themselves run INSIDE the task
# container, and their dependencies are installed by each task's
# environment/Dockerfile -- see the "Grader runtime" block there, and
# validate_task.py check C12 which enforces its presence. They are listed at the
# bottom for reference, not installed here.

# --- the benchmark runner --------------------------------------------------
# Pinned, not floating: a runner upgrade changes trial semantics (verifier
# modes, artifact collection, network policy), so it is a deliberate act.
harbor==0.20.0

# --- Claude Code OAuth bridge (claude_code/) -------------------------------
# An Anthropic-compatible proxy backed by subscription OAuth, so agent and
# grader traffic can run without an API key.
fastapi==0.141.1
uvicorn==0.52.1

# --- harness scripts -------------------------------------------------------
# httpx  : bridge forwarding, probe_grader.py, and the graders when run locally
# pyyaml : workflows.yaml parsing in score.py and run_workflows.py
httpx==0.28.1
pyyaml==6.0.3
